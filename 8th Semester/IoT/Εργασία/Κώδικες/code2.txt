#define BLYNK_PRINT Serial
#define BLYNK_TEMPLATE_ID "TMPL4ejA72qMA"
#define BLYNK_TEMPLATE_NAME "Traffic Light Controller 2"

#include <WiFi.h>
#include <WiFiClient.h>
#include <BlynkSimpleEsp32.h>

char auth[] = "OhCx5HUskMDHEgUCqMhRNVbUsK9LdIBD";
char ssid[] = "Wokwi-GUEST";
char pass[] = "";

#define RED_LED 2
#define YELLOW_LED 4
#define GREEN_LED 5
#define BLUE_BUTTON 12

bool lastButtonState = HIGH; // Αρχική κατάσταση του κουμπιού

// Μεταβλητές για τον έλεγχο του χρόνου
unsigned long lastDebounceTime = 0; // Τελευταία φορά που άλλαξε η κατάσταση του κουμπιού
unsigned long debounceDelay = 50;   // Χρόνος ελάχιστου διακοπής για την αποφυγή θορύβου

BlynkTimer timer;

void readBlueButton() {
  // Ο κώδικας διαβάζει την κατάσταση του κουμπιού από το Blynk
  int buttonState = digitalRead(BLUE_BUTTON);

  // Ελέγχει αν η κατάσταση του κουμπιού έχει αλλάξει
  if (buttonState != lastButtonState) {
    // Αν έχει αλλάξει, αποθηκεύει την ώρα της αλλαγής
    lastDebounceTime = millis();
  }

  // Ελέγχει αν έχει περάσει αρκετός χρόνος από την τελευταία αλλαγή
  if ((millis() - lastDebounceTime) > debounceDelay) {
    // Αν έχει περάσει αρκετός χρόνος, ελέγχει αν η κατάσταση του κουμπιού έχει αλλάξει ξανά
    if (buttonState != lastButtonState) {
      // Αν έχει αλλάξει ξανά, ελέγχει αν το κουμπί πατήθηκε (είναι LOW)
      if (buttonState == LOW) {
        // Αν το κουμπί πατήθηκε, καλεί την συνάρτηση για αλλαγή των φώτων
        changeLightsManually();
      }
    }
  }

  // Αποθηκεύει την τρέχουσα κατάσταση του κουμπιού για να την χρησιμοποιήσει στον επόμενο κύκλο ελέγχου
  lastButtonState = buttonState;
}

void setup() {
  pinMode(RED_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(BLUE_BUTTON, INPUT_PULLUP);
  Serial.begin(9600);
  
  WiFi.begin(ssid, pass);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("");
  Serial.println("WiFi connected");
  
  Blynk.begin(auth, ssid, pass);

  //timer.setInterval(100L, readBlueButton); // Καθορίζουμε το χρονικό διάστημα για τον έλεγχο του μπλε κουμπιού
}

void loop() {
  Blynk.run();
  timer.run();
}

void changeLightsManually() {
  static int state = 0;

  switch(state) {
    case 0:
      Blynk.virtualWrite(RED_LED, HIGH);
      Blynk.virtualWrite(YELLOW_LED, LOW);
      Blynk.virtualWrite(GREEN_LED, LOW);
      state = 1;
      break;
    case 1:
      Blynk.virtualWrite(RED_LED, HIGH);
      Blynk.virtualWrite(YELLOW_LED, HIGH);
      Blynk.virtualWrite(GREEN_LED, LOW);
      state = 2;
      break;
    case 2:
      Blynk.virtualWrite(RED_LED, LOW);
      Blynk.virtualWrite(YELLOW_LED, LOW);
      Blynk.virtualWrite(GREEN_LED, HIGH);
      state = 3;
      break;
    case 3:
      Blynk.virtualWrite(RED_LED, LOW);
      Blynk.virtualWrite(YELLOW_LED, HIGH);
      Blynk.virtualWrite(GREEN_LED, LOW);
      state = 0;
      break;
  }
}
