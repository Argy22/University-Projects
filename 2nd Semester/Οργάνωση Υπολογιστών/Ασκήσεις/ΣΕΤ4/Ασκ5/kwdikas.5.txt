int buttonPin1 = 2;
int buttonPin2 = 3;
int Ledpin = 8;
int A = 0;
int D = 0;

void setup() {
pinMode(Ledpin, OUTPUT);
pinMode(A, INPUT);
pinMode(D, INPUT);
}

void loop(){
A = digitalRead(buttonPin1);
D = digitalRead(buttonPin2);

 if(A||D)
  {
  digitalWrite(Ledpin,HIGH); 
  }
  else
  digitalWrite(Ledpin,LOW);
}
